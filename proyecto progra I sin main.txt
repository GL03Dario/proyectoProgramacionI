colecciones/Nodo.h
#ifndef NODO_H
#define NODO_H


template <typename T>
class Nodo {
private:
	T* dato;
	Nodo<T>* siguiente;

public:
	explicit Nodo(T* dato) : dato(dato), siguiente(nullptr) {}

	Nodo(const Nodo&) = delete;
	Nodo& operator=(const Nodo&) = delete;

	T* getDato() const { return dato; }
	Nodo<T>* getSiguiente() const { return siguiente; }
	void setSiguiente(Nodo<T>* nuevoSiguiente) { siguiente = nuevoSiguiente; }
};

#endif

colecciones/Lista.h
#ifndef LISTA_H
#define LISTA_H

#include "colecciones/Nodo.h"
#include <stdexcept>
#include <string>


template <typename T>
class Lista {
private:
	Nodo<T>* cabeza;
	Nodo<T>* cola;
	int tamano;

public:
	Lista() : cabeza(nullptr), cola(nullptr), tamano(0) {}

	~Lista() { limpiar(); }

	Lista(const Lista&) = delete;
	Lista& operator=(const Lista&) = delete;

	
	void agregar(T* elemento) {
		if (elemento == nullptr) {
			throw std::invalid_argument(
				"No se puede agregar un elemento nulo a la lista.");
		}

		Nodo<T>* nuevo = new Nodo<T>(elemento);

		if (cola == nullptr) {
			cabeza = nuevo;
		} else {
			cola->setSiguiente(nuevo);
		}

		cola = nuevo;
		++tamano;
	}

	
	bool eliminar(const T* elemento) {
		Nodo<T>* anterior = nullptr;
		Nodo<T>* actual = cabeza;

		while (actual != nullptr) {
			if (actual->getDato() == elemento) {
				Nodo<T>* siguiente = actual->getSiguiente();

				if (anterior == nullptr) {
					cabeza = siguiente;
				} else {
					anterior->setSiguiente(siguiente);
				}

				if (actual == cola) {
					cola = anterior;
				}

				delete actual->getDato();
				delete actual;
				--tamano;
				return true;
			}

			anterior = actual;
			actual = actual->getSiguiente();
		}

		return false;
	}

	T* obtener(int indice) const {
		if (indice < 0 || indice >= tamano) {
			throw std::out_of_range(
				"Indice fuera de rango: " + std::to_string(indice));
		}

		Nodo<T>* actual = cabeza;

		for (int i = 0; i < indice; ++i) {
			actual = actual->getSiguiente();
		}

		return actual->getDato();
	}

	
	template <typename Predicado>
	T* buscar(Predicado predicado) const {
		for (Nodo<T>* actual = cabeza;
			 actual != nullptr;
			 actual = actual->getSiguiente()) {

			if (predicado(actual->getDato())) {
				return actual->getDato();
			}
		}

		return nullptr;
	}

	
	Nodo<T>* primero() const { return cabeza; }

	int getTamano() const { return tamano; }
	bool estaVacia() const { return tamano == 0; }

	void limpiar() {
		Nodo<T>* actual = cabeza;

		while (actual != nullptr) {
			Nodo<T>* siguiente = actual->getSiguiente();
			delete actual->getDato();
			delete actual;
			actual = siguiente;
		}

		cabeza = nullptr;
		cola = nullptr;
		tamano = 0;
	}
};

#endif

#include "modelo/Empleado.h"
#include <stdexcept>

Empleado::Empleado(
	const std::string& identificacion,
	const std::string& nombre,
	double salarioBase,
	double horasExtra,
	double valorHoraExtra,
	int turnosNocturnos,
	double bonificacionTurnoNocturno,
	int guardias,
	double bonificacionGuardia,
	double tasaDeduccionLegal)

	: identificacion(identificacion),
	  nombre(nombre),
	  salarioBase(salarioBase),
	  horasExtra(horasExtra),
	  valorHoraExtra(valorHoraExtra),
	  turnosNocturnos(turnosNocturnos),
	  bonificacionTurnoNocturno(bonificacionTurnoNocturno),
	  guardias(guardias),
	  bonificacionGuardia(bonificacionGuardia),
	  tasaDeduccionLegal(tasaDeduccionLegal),
	  idDepartamento(),
	  codigoPuesto() {

	if (identificacion.empty())
		throw std::invalid_argument(
			"La identificacion no puede estar vacia.");

	if (nombre.empty())
		throw std::invalid_argument(
			"El nombre no puede estar vacio.");

	validarMonto(salarioBase, "salario base");
	validarMonto(horasExtra, "horas extra");
	validarMonto(valorHoraExtra, "valor de hora extra");

	validarMonto(
		bonificacionTurnoNocturno,
		"bonificacion nocturna");

	validarMonto(
		bonificacionGuardia,
		"bonificacion por guardia");

	validarCantidad(turnosNocturnos, "turnos nocturnos");
	validarCantidad(guardias, "guardias");

	if (tasaDeduccionLegal < 0.0 ||
		tasaDeduccionLegal > 1.0) {

		throw std::invalid_argument(
			"La tasa de deduccion debe estar entre 0 y 1.");
	}
}

Empleado::~Empleado() = default;

void Empleado::validarMonto(
	double valor, const std::string& campo) {

	if (valor < 0.0) {
		throw std::invalid_argument(
			"El valor de " + campo + " no puede ser negativo.");
	}
}

void Empleado::validarCantidad(
	int valor, const std::string& campo) {

	if (valor < 0) {
		throw std::invalid_argument(
			"La cantidad de " + campo + " no puede ser negativa.");
	}
}

double Empleado::calcularPagoHorasExtra() const {
	return horasExtra * valorHoraExtra;
}

double Empleado::calcularBonificacionNocturna() const {
	return turnosNocturnos * bonificacionTurnoNocturno;
}

double Empleado::calcularBonificacionGuardias() const {
	return guardias * bonificacionGuardia;
}

double Empleado::calcularSalarioBruto() const {

	return salarioBase
		 + calcularPagoHorasExtra()
		 + calcularBonificacionNocturna()
		 + calcularBonificacionGuardias()
		 + calcularBonificacionPuesto();
}

double Empleado::calcularDeduccionLegal() const {
	return calcularSalarioBruto() * tasaDeduccionLegal;
}

double Empleado::calcularSalarioNeto() const {
	return calcularSalarioBruto()
		 - calcularDeduccionLegal();
}

const std::string& Empleado::getIdentificacion() const {
	return identificacion;
}

const std::string& Empleado::getNombre() const {
	return nombre;
}

double Empleado::getSalarioBase() const {
	return salarioBase;
}

double Empleado::getHorasExtra() const {
	return horasExtra;
}

double Empleado::getValorHoraExtra() const {

}

int Empleado::getTurnosNocturnos() const {

}

double Empleado::getBonificacionTurnoNocturno() const {

int Empleado::getGuardias() const {
	return guardias;
}

double Empleado::getBonificacionGuardia() const {
	return bonificacionGuardia;
}

double Empleado::getTasaDeduccionLegal() const {
	return tasaDeduccionLegal;
}

const std::string& Empleado::getIdDepartamento() const {
	return idDepartamento;
}

const std::string& Empleado::getCodigoPuesto() const {
	return codigoPuesto;
}

void Empleado::setNombre(const std::string& valor) {
	if (valor.empty())
		throw std::invalid_argument(
			"El nombre no puede estar vacio.");

	nombre = valor;
}

	const std::string& codigoPuesto,
	const std::string& nombrePuesto) {

	if (existePuesto(codigoPuesto)) {
		throw std::invalid_argument(
			"Ya existe un puesto con el codigo " + codigoPuesto +
			" en el departamento " + codigo + ".");
	}

	puestos.agregar(new Puesto(codigoPuesto, nombrePuesto));
}

bool Departamento::eliminarPuesto(const std::string& codigoPuesto) {
	Puesto* puesto = buscarPuesto(codigoPuesto);

	if (puesto == nullptr) {
		return false;
	}

	return puestos.eliminar(puesto);
}

Puesto* Departamento::buscarPuesto(
	const std::string& codigoPuesto) const {

	return puestos.buscar([&](const Puesto* puesto) {
		return puesto->getCodigo() == codigoPuesto;
	});
}

bool Departamento::existePuesto(const std::string& codigoPuesto) const {
	return buscarPuesto(codigoPuesto) != nullptr;
}

const Lista<Puesto>& Departamento::getPuestos() const {
	return puestos;
}

	});

void FabricaEmpleado::registrar(
	const std::string& tipo,
	const std::string& etiquetaDetalle,
	CreadorEmpleado creador) {

	if (tipo.empty()) {
		throw std::invalid_argument(
			"El tipo de empleado no puede estar vacio.");
	}

	if (creador == nullptr) {
		throw std::invalid_argument(
			"El creador del tipo " + tipo + " no puede ser nulo.");
	}

	if (buscarRegistro(tipo) != nullptr) {
		throw std::invalid_argument(
			"El tipo de empleado " + tipo + " ya esta registrado.");
	}

	registros.agregar(new RegistroCreador{tipo, etiquetaDetalle, creador});
}

bool FabricaEmpleado::soporta(const std::string& tipo) const {
	return buscarRegistro(tipo) != nullptr;
}

std::string FabricaEmpleado::obtenerEtiquetaDetalle(
	const std::string& tipo) const {

	const RegistroCreador* registro = buscarRegistro(tipo);

	if (registro == nullptr) {
		return "Detalle del puesto";
	}

	return registro->etiquetaDetalle;
}

Empleado* FabricaEmpleado::crear(const DatosEmpleado& datos) const {
	const RegistroCreador* registro = buscarRegistro(datos.tipo);

	if (registro == nullptr) {
		throw std::invalid_argument(
			"Tipo de empleado desconocido: " + datos.tipo);
	}

	Empleado* empleado = registro->creador(datos);
	empleado->setIdDepartamento(datos.idDepartamento);
	empleado->setCodigoPuesto(datos.codigoPuesto);

	return empleado;
}

const Lista<RegistroCreador>& FabricaEmpleado::getRegistros() const {
	return registros;
}

#include "persistencia/AlmacenamientoArchivos.h"
#include <filesystem>
#include <fstream>
#include <functional>
#include <iomanip>
#include <locale>
#include <sstream>
#include <stdexcept>
#include <system_error>
#include <vector>

namespace {

	const char SEPARADOR = '|';

	std::vector<std::string> dividir(const std::string& linea) {
		std::vector<std::string> campos;
		std::string actual;

		for (char c : linea) {
			if (c == SEPARADOR) {
				campos.push_back(actual);
				actual.clear();
			} else {
				actual += c;
			}
		}

		campos.push_back(actual);
		return campos;
	}

	
	std::string limpiarCampo(const std::string& texto) {
		std::string resultado = texto;

		for (char& c : resultado) {
			if (c == SEPARADOR) {
				c = '/';
			} else if (c == '\n' || c == '\r') {
				c = ' ';
			}
		}

		return resultado;
	}

	std::string aTexto(double valor) {
		std::ostringstream flujo;
		flujo.imbue(std::locale::classic());
		flujo << std::setprecision(15) << valor;
		return flujo.str();
	}

	double aDecimal(const std::string& texto, const std::string& campo) {
		std::istringstream flujo(texto);
		flujo.imbue(std::locale::classic());

		double valor = 0.0;
		flujo >> valor;

		if (flujo.fail()) {
			throw std::invalid_argument(
				"Valor decimal invalido en " + campo + ": '" + texto + "'.");
		}

		flujo >> std::ws;

		if (!flujo.eof()) {
			throw std::invalid_argument(
				"Valor decimal invalido en " + campo + ": '" + texto + "'.");
		}

		return valor;
	}

	int aEntero(const std::string& texto, const std::string& campo) {
		std::size_t posicion = 0;
		int valor = 0;

		try {
			valor = std::stoi(texto, &posicion);
		} catch (const std::exception&) {
			throw std::invalid_argument(
				"Valor entero invalido en " + campo + ": '" + texto + "'.");
		}

		if (posicion != texto.size()) {
			throw std::invalid_argument(
				"Valor entero invalido en " + campo + ": '" + texto + "'.");
		}

		return valor;
	}

	bool aBooleano(const std::string& texto, const std::string& campo) {
		if (texto == "1") {
			return true;
		}

		if (texto == "0") {
			return false;
		}

		throw std::invalid_argument(
			"Valor booleano invalido en " + campo + ": '" + texto + "'.");
	}

	void exigirCampos(
		const std::vector<std::string>& campos, std::size_t esperados) {

		if (campos.size() != esperados) {
			throw std::invalid_argument(
				"Se esperaban " + std::to_string(esperados) +
				" campos y se encontraron " +
				std::to_string(campos.size()) + ".");
		}
	}

	
	void procesarLineas(
		const std::string& ruta,
		const std::function<void(const std::string&)>& procesador) {

		std::ifstream archivo(ruta);

		if (!archivo.is_open()) {
			return;
		}

		std::string linea;
		int numeroLinea = 0;

		while (std::getline(archivo, linea)) {
			++numeroLinea;

			if (!linea.empty() && linea.back() == '\r') {
				linea.pop_back();
			}

			if (linea.empty() || linea[0] == '#') {
				continue;
			}

			try {
				procesador(linea);
			} catch (const std::exception& e) {
				throw std::runtime_error(
					ruta + ", linea " + std::to_string(numeroLinea) +
					": " + e.what());
			}
		}
	}

	
	void escribirArchivo(
		const std::string& ruta, const std::string& contenido) {

		const std::string temporal = ruta + ".tmp";

		{
			std::ofstream salida(temporal, std::ios::trunc);

			if (!salida.is_open()) {
				throw std::runtime_error(
					"No se pudo abrir el archivo " + temporal);
			}

			salida << contenido;
			salida.flush();

			if (!salida) {
				throw std::runtime_error(
					"Error al escribir el archivo " + temporal);
			}
		}

		std::error_code error;
		std::filesystem::rename(temporal, ruta, error);

		if (error) {
			throw std::runtime_error(
				"No se pudo actualizar " + ruta + ": " + error.message());
		}
	}

}

void AlmacenamientoArchivos::cargarEmpleados(Lista<Empleado>& destino) {
	destino.limpiar();

	procesarLineas(rutaEmpleados, [&](const std::string& linea) {
		std::vector<std::string> c = dividir(linea);
		exigirCampos(c, 15);

		DatosEmpleado datos;
		datos.tipo = c[0];
		datos.identificacion = c[1];
		datos.nombre = c[2];
		datos.salarioBase = aDecimal(c[3], "salario base");
		datos.horasExtra = aDecimal(c[4], "horas extra");
		datos.valorHoraExtra = aDecimal(c[5], "valor hora extra");
		datos.turnosNocturnos = aEntero(c[6], "turnos nocturnos");
		datos.bonificacionTurnoNocturno = aDecimal(c[7], "bonificacion nocturna");
		datos.guardias = aEntero(c[8], "guardias");
		datos.bonificacionGuardia = aDecimal(c[9], "bonificacion guardia");
		datos.tasaDeduccionLegal = aDecimal(c[10], "tasa de deduccion");
		datos.idDepartamento = c[11];
		datos.codigoPuesto = c[12];
		datos.detallePuesto = c[13];
		datos.bonificacionPuesto = aDecimal(c[14], "bonificacion del puesto");

		destino.agregar(fabrica.crear(datos));
	});
}

void AlmacenamientoArchivos::guardarEmpleados(const Lista<Empleado>& origen) {
	std::ostringstream contenido;

	for (Nodo<Empleado>* nodo = origen.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		const Empleado* e = nodo->getDato();

		contenido
			<< limpiarCampo(e->obtenerTipo()) << SEPARADOR
			<< limpiarCampo(e->getIdentificacion()) << SEPARADOR
			<< limpiarCampo(e->getNombre()) << SEPARADOR
			<< aTexto(e->getSalarioBase()) << SEPARADOR
			<< aTexto(e->getHorasExtra()) << SEPARADOR
			<< aTexto(e->getValorHoraExtra()) << SEPARADOR
			<< e->getTurnosNocturnos() << SEPARADOR
			<< aTexto(e->getBonificacionTurnoNocturno()) << SEPARADOR
			<< e->getGuardias() << SEPARADOR
			<< aTexto(e->getBonificacionGuardia()) << SEPARADOR
			<< aTexto(e->getTasaDeduccionLegal()) << SEPARADOR
			<< limpiarCampo(e->getIdDepartamento()) << SEPARADOR
			<< limpiarCampo(e->getCodigoPuesto()) << SEPARADOR
			<< limpiarCampo(e->obtenerDetallePuesto()) << SEPARADOR
			<< aTexto(e->calcularBonificacionPuesto())
			<< '\n';
	}

	escribirArchivo(rutaEmpleados, contenido.str());
}



void AlmacenamientoArchivos::cargarDepartamentos(
	Lista<Departamento>& destino) {

	destino.limpiar();

	procesarLineas(rutaDepartamentos, [&](const std::string& linea) {
		std::vector<std::string> c = dividir(linea);

		if (c[0] == "D") {
			exigirCampos(c, 3);
			destino.agregar(new Departamento(c[1], c[2]));
		} else if (c[0] == "P") {
			exigirCampos(c, 4);

			Departamento* departamento = destino.buscar(
				[&](const Departamento* d) {
					return d->getCodigo() == c[1];
				});

			if (departamento == nullptr) {
				throw std::invalid_argument(
					"El puesto " + c[2] + " hace referencia a un "
					"departamento inexistente: " + c[1]);
			}

			departamento->agregarPuesto(c[2], c[3]);
		} else {
			throw std::invalid_argument(
				"Tipo de registro desconocido: '" + c[0] + "'.");
		}
	});
}

void AlmacenamientoArchivos::guardarDepartamentos(
	const Lista<Departamento>& origen) {

	std::ostringstream contenido;

	for (Nodo<Departamento>* nodo = origen.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		const Departamento* d = nodo->getDato();

		contenido << "D" << SEPARADOR
				  << limpiarCampo(d->getCodigo()) << SEPARADOR
				  << limpiarCampo(d->getNombre()) << '\n';

		const Lista<Puesto>& puestos = d->getPuestos();

		for (Nodo<Puesto>* np = puestos.primero();
			 np != nullptr;
			 np = np->getSiguiente()) {

			contenido << "P" << SEPARADOR
					  << limpiarCampo(d->getCodigo()) << SEPARADOR
					  << limpiarCampo(np->getDato()->getCodigo()) << SEPARADOR
					  << limpiarCampo(np->getDato()->getNombre()) << '\n';
		}
	}

	escribirArchivo(rutaDepartamentos, contenido.str());
}



void AlmacenamientoArchivos::cargarComprobantes(
	Lista<ComprobantePago>& destino,
	const Lista<Empleado>& empleados,
	const Lista<Departamento>& departamentos) {

	destino.limpiar();

	procesarLineas(rutaPlanilla, [&](const std::string& linea) {
		std::vector<std::string> c = dividir(linea);
		exigirCampos(c, 18);

		DesglosePago desglose;
		desglose.salarioBase = aDecimal(c[10], "salario base");
		desglose.pagoHorasExtra = aDecimal(c[11], "pago horas extra");
		desglose.bonificacionNocturna = aDecimal(c[12], "bonificacion nocturna");
		desglose.bonificacionGuardias = aDecimal(c[13], "bonificacion guardias");
		desglose.bonificacionPuesto = aDecimal(c[14], "bonificacion puesto");
		desglose.salarioBruto = aDecimal(c[15], "salario bruto");
		desglose.deduccionLegal = aDecimal(c[16], "deduccion legal");
		desglose.salarioNeto = aDecimal(c[17], "salario neto");

		
		const Empleado* empleado = empleados.buscar(
			[&](const Empleado* e) {
				return e->getIdentificacion() == c[5];
			});

		const Departamento* departamento = nullptr;

		if (!c[8].empty()) {
			departamento = departamentos.buscar(
				[&](const Departamento* d) {
					return d->getCodigo() == c[8];
				});
		}

		destino.agregar(new ComprobantePago(
			aEntero(c[0], "numero de comprobante"),
			c[1],
			c[2],
			aBooleano(c[3], "estado de pago"),
			c[4],
			c[5],
			c[6],
			c[7],
			c[8],
			c[9],
			desglose,
			empleado,
			departamento));
	});
}

void AlmacenamientoArchivos::guardarComprobantes(
	const Lista<ComprobantePago>& origen) {

	std::ostringstream contenido;

	for (Nodo<ComprobantePago>* nodo = origen.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		const ComprobantePago* p = nodo->getDato();
		const DesglosePago& d = p->getDesglose();

		contenido
			<< p->getNumero() << SEPARADOR
			<< limpiarCampo(p->getPeriodo()) << SEPARADOR
			<< limpiarCampo(p->getFechaEmision()) << SEPARADOR
			<< (p->estaPagado() ? "1" : "0") << SEPARADOR
			<< limpiarCampo(p->getFechaPago()) << SEPARADOR
			<< limpiarCampo(p->getIdEmpleado()) << SEPARADOR
			<< limpiarCampo(p->getNombreEmpleado()) << SEPARADOR
			<< limpiarCampo(p->getTipoEmpleado()) << SEPARADOR
			<< limpiarCampo(p->getCodigoDepartamento()) << SEPARADOR
			<< limpiarCampo(p->getNombreDepartamento()) << SEPARADOR
			<< aTexto(d.salarioBase) << SEPARADOR
			<< aTexto(d.pagoHorasExtra) << SEPARADOR
			<< aTexto(d.bonificacionNocturna) << SEPARADOR
			<< aTexto(d.bonificacionGuardias) << SEPARADOR
			<< aTexto(d.bonificacionPuesto) << SEPARADOR
			<< aTexto(d.salarioBruto) << SEPARADOR
			<< aTexto(d.deduccionLegal) << SEPARADOR
			<< aTexto(d.salarioNeto)
			<< '\n';
	}

	escribirArchivo(rutaPlanilla, contenido.str());
}

	const std::string& codigo, const std::string& nuevoNombre) {

	if (departamento == nullptr) {
		throw std::invalid_argument(
			"No existe un departamento con el codigo " + codigo + ".");
	}

	departamento->setNombre(nuevoNombre);
	guardar();
}

	Departamento* departamento = buscar(codigo);

	if (departamento == nullptr) {

	departamentos.eliminar(departamento);

		throw std::invalid_argument(

	const std::string& codigoDepartamento,
	const std::string& codigoPuesto) {

	Departamento* departamento = buscar(codigoDepartamento);

	if (departamento == nullptr) {
		throw std::invalid_argument(
			"No existe un departamento con el codigo " +

	const FabricaEmpleado& fabrica,

}

void ControladorEmpleados::validarAsignacion(
	const std::string& idDepartamento,
	const std::string& codigoPuesto) const {

	if (idDepartamento.empty()) {
		if (!codigoPuesto.empty()) {
			throw std::invalid_argument(
				"No se puede asignar un puesto sin departamento.");
		}

		return;
	}

	if (departamentos.buscar(idDepartamento) == nullptr) {
		throw std::invalid_argument(
			"No existe un departamento con el codigo " +
			idDepartamento + ".");
	}

	if (!codigoPuesto.empty() &&
		!departamentos.existePuesto(idDepartamento, codigoPuesto)) {

		throw std::invalid_argument(
			"No existe el puesto " + codigoPuesto +
			" en el departamento " + idDepartamento + ".");
	}
}

Empleado* ControladorEmpleados::registrar(const DatosEmpleado& datos) {
	if (buscar(datos.identificacion) != nullptr) {
		throw std::invalid_argument(
			"Ya existe un empleado con la identificacion " +
			datos.identificacion + ".");
	}

	validarAsignacion(datos.idDepartamento, datos.codigoPuesto);

	Empleado* empleado = fabrica.crear(datos);
	empleados.agregar(empleado);
	guardar();

	return empleado;
}

Empleado* ControladorEmpleados::buscar(
	const std::string& identificacion) const {

	return empleados.buscar([&](const Empleado* e) {
		return e->getIdentificacion() == identificacion;
	});
}

void ControladorEmpleados::eliminar(const std::string& identificacion) {
	Empleado* empleado = buscar(identificacion);

	if (empleado == nullptr) {
		throw std::invalid_argument(
			"No existe un empleado con la identificacion " +
			identificacion + ".");
	}

	empleados.eliminar(empleado);
	guardar();
}

void ControladorEmpleados::asignarPuesto(
	const std::string& identificacion,
	const std::string& idDepartamento,
	const std::string& codigoPuesto) {

	Empleado* empleado = buscar(identificacion);

	if (empleado == nullptr) {
		throw std::invalid_argument(
			"No existe un empleado con la identificacion " +
			identificacion + ".");
	}

	validarAsignacion(idDepartamento, codigoPuesto);

	empleado->setIdDepartamento(idDepartamento);
	empleado->setCodigoPuesto(codigoPuesto);
	guardar();
}

void ControladorEmpleados::desasignarDepartamento(
	const std::string& idDepartamento) {

	bool huboCambios = false;

	for (Nodo<Empleado>* nodo = empleados.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		Empleado* empleado = nodo->getDato();

		if (empleado->getIdDepartamento() == idDepartamento) {
			empleado->setIdDepartamento("");
			empleado->setCodigoPuesto("");
			huboCambios = true;
		}
	}

	if (huboCambios) {
		guardar();
	}
}

void ControladorEmpleados::desasignarPuesto(
	const std::string& idDepartamento,
	const std::string& codigoPuesto) {

	bool huboCambios = false;

	for (Nodo<Empleado>* nodo = empleados.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		Empleado* empleado = nodo->getDato();

		if (empleado->getIdDepartamento() == idDepartamento &&
			empleado->getCodigoPuesto() == codigoPuesto) {

			empleado->setCodigoPuesto("");
			huboCambios = true;
		}
	}

	if (huboCambios) {
		guardar();
	}
}

		if (nodo->getDato()->getIdDepartamento() == idDepartamento) {

}

void ServicioPlanilla::validarPeriodo(const std::string& periodo) {
	bool valido = periodo.size() == 7 && periodo[4] == '-';

	for (std::size_t i = 0; valido && i < periodo.size(); ++i) {
		if (i != 4 &&
			!std::isdigit(static_cast<unsigned char>(periodo[i]))) {

			valido = false;
		}
	}

	if (valido) {
		int mes = std::stoi(periodo.substr(5, 2));
		valido = mes >= 1 && mes <= 12;
	}

	if (!valido) {
		throw std::invalid_argument(
			"El periodo debe tener el formato AAAA-MM (ejemplo: 2026-11).");
	}
}

std::string ServicioPlanilla::fechaActual() {
	std::time_t ahora = std::time(nullptr);
	std::tm* local = std::localtime(&ahora);

	if (local == nullptr) {
		return "0000-00-00";
	}

	char buffer[16];
	std::strftime(buffer, sizeof(buffer), "%Y-%m-%d", local);

	return buffer;
}

int ServicioPlanilla::siguienteNumero() const {
	int mayor = 0;

	for (Nodo<ComprobantePago>* nodo = comprobantes.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		if (nodo->getDato()->getNumero() > mayor) {
			mayor = nodo->getDato()->getNumero();
		}
	}

	return mayor + 1;
}

bool ServicioPlanilla::existeComprobante(
	const std::string& idEmpleado, const std::string& periodo) const {

	return comprobantes.buscar([&](const ComprobantePago* c) {
		return c->getIdEmpleado() == idEmpleado &&
			   c->getPeriodo() == periodo;
	}) != nullptr;
}

ComprobantePago* ServicioPlanilla::emitir(
	const Empleado* empleado, const std::string& periodo) {

	const Departamento* departamento = nullptr;

	if (!empleado->getIdDepartamento().empty()) {
		departamento = departamentos.buscar(empleado->getIdDepartamento());
	}

	ComprobantePago* comprobante = new ComprobantePago(
		siguienteNumero(),
		periodo,
		fechaActual(),
		empleado,
		departamento);

	comprobantes.agregar(comprobante);

	return comprobante;
}

ComprobantePago* ServicioPlanilla::generarComprobante(
	const std::string& idEmpleado, const std::string& periodo) {

	validarPeriodo(periodo);

	const Empleado* empleado = empleados.buscar(idEmpleado);

	if (empleado == nullptr) {
		throw std::invalid_argument(
			"No existe un empleado con la identificacion " +
			idEmpleado + ".");
	}

	if (existeComprobante(idEmpleado, periodo)) {
		throw std::logic_error(
			"El empleado " + idEmpleado +
			" ya tiene un comprobante en el periodo " + periodo + ".");
	}

	ComprobantePago* comprobante = emitir(empleado, periodo);
	guardar();

	return comprobante;
}

int ServicioPlanilla::generarPlanilla(const std::string& periodo) {
	validarPeriodo(periodo);

	int generados = 0;

	for (Nodo<Empleado>* nodo = empleados.getEmpleados().primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		const Empleado* empleado = nodo->getDato();

		if (!existeComprobante(empleado->getIdentificacion(), periodo)) {
			emitir(empleado, periodo);
			++generados;
		}
	}

	if (generados > 0) {
		guardar();
	}

	return generados;
}

void ServicioPlanilla::procesarPago(int numeroComprobante) {
	ComprobantePago* comprobante = buscarComprobante(numeroComprobante);

	if (comprobante == nullptr) {
		throw std::invalid_argument(
			"No existe el comprobante numero " +
			std::to_string(numeroComprobante) + ".");
	}

	comprobante->marcarPagado(fechaActual());
	guardar();
}

int ServicioPlanilla::procesarPagosPeriodo(const std::string& periodo) {
	validarPeriodo(periodo);

	int pagados = 0;
	const std::string fecha = fechaActual();

	for (Nodo<ComprobantePago>* nodo = comprobantes.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		ComprobantePago* comprobante = nodo->getDato();

		if (comprobante->getPeriodo() == periodo &&
			!comprobante->estaPagado()) {

			comprobante->marcarPagado(fecha);
			++pagados;
		}
	}

	if (pagados > 0) {
		guardar();
	}

	return pagados;
}

ComprobantePago* ServicioPlanilla::buscarComprobante(int numero) const {
	return comprobantes.buscar([&](const ComprobantePago* c) {
		return c->getNumero() == numero;
	});
}

namespace {

	std::string recortar(const std::string& texto) {
		std::size_t inicio = 0;
		std::size_t fin = texto.size();

		while (inicio < fin &&
			   std::isspace(static_cast<unsigned char>(texto[inicio]))) {
			++inicio;
		}

		while (fin > inicio &&
			   std::isspace(static_cast<unsigned char>(texto[fin - 1]))) {
			--fin;
		}

		return texto.substr(inicio, fin - inicio);
	}

	bool esPeriodoValido(const std::string& periodo) {
		if (periodo.size() != 7 || periodo[4] != '-') {
			return false;
		}

		for (std::size_t i = 0; i < periodo.size(); ++i) {
			if (i != 4 &&
				!std::isdigit(static_cast<unsigned char>(periodo[i]))) {
				return false;
			}
		}

		int mes = std::stoi(periodo.substr(5, 2));
		return mes >= 1 && mes <= 12;
	}

}

std::string EntradaConsola::leerLinea(const std::string& mensaje) {
	std::cout << mensaje;

	std::string linea;

	if (!std::getline(std::cin, linea)) {
		throw EntradaFinalizada();
	}

	return recortar(linea);
}

std::string EntradaConsola::leerTexto(
	const std::string& mensaje, bool permitirVacio) {

	while (true) {
		std::string texto = leerLinea(mensaje);

		if (texto.empty() && !permitirVacio) {
			std::cout << "  Error: el valor no puede estar vacío.\n";
			continue;
		}

		if (texto.find('|') != std::string::npos) {
			std::cout << "  Error: el carácter '|' no está permitido.\n";
			continue;
		}

		return texto;
	}
}

int EntradaConsola::leerEntero(
	const std::string& mensaje, int minimo, int maximo) {

	while (true) {
		std::string texto = leerLinea(mensaje);
		std::size_t posicion = 0;
		int valor = 0;

		try {
			valor = std::stoi(texto, &posicion);
		} catch (const std::exception&) {
			std::cout << "  Error: debe ingresar un número entero.\n";
			continue;
		}

		if (posicion != texto.size()) {
			std::cout << "  Error: debe ingresar un número entero.\n";
			continue;
		}

		if (valor < minimo || valor > maximo) {
			std::cout << "  Error: el valor debe estar entre "
					  << minimo << " y " << maximo << ".\n";
			continue;
		}

		return valor;
	}
}

double EntradaConsola::leerDecimal(
	const std::string& mensaje, double minimo, double maximo) {

	while (true) {
		std::string texto = leerLinea(mensaje);

		
		for (char& c : texto) {
			if (c == ',') {
				c = '.';
			}
		}

		std::istringstream flujo(texto);
		flujo.imbue(std::locale::classic());

		double valor = 0.0;
		flujo >> valor;

		if (flujo.fail()) {
			std::cout << "  Error: debe ingresar un número.\n";
			continue;
		}

		flujo >> std::ws;

		if (!flujo.eof()) {
			std::cout << "  Error: debe ingresar un número.\n";
			continue;
		}

		if (valor < minimo || valor > maximo) {
			std::cout << "  Error: el valor debe estar entre "
					  << minimo << " y " << maximo << ".\n";
			continue;
		}

		return valor;
	}
}

bool EntradaConsola::confirmar(const std::string& mensaje) {
	while (true) {
		std::string texto = leerLinea(mensaje + " (s/n): ");

		for (char& c : texto) {
			c = static_cast<char>(
				std::tolower(static_cast<unsigned char>(c)));
		}

		if (texto == "s" || texto == "si") {
			return true;
		}

		if (texto == "n" || texto == "no") {
			return false;
		}

		std::cout << "  Error: responda 's' o 'n'.\n";
	}
}

std::string EntradaConsola::leerPeriodo(const std::string& mensaje) {
	while (true) {
		std::string texto = leerLinea(mensaje);

		if (esPeriodoValido(texto)) {
			return texto;
		}

		std::cout << "  Error: use el formato AAAA-MM (ejemplo: 2026-11).\n";
	}
}

void EntradaConsola::pausar() {
	leerLinea("\nPresione Enter para continuar...");
}

		const Departamento* d = departamentos.buscar(e->getIdDepartamento());

		nombreDepartamento = e->getIdDepartamento();

		if (d != nullptr) {
			nombreDepartamento += " - " + d->getNombre();

	std::cout << "\n  Tipo:            " << e->obtenerTipo()
			  << "\n  Identificación:  " << e->getIdentificacion()
			  << "\n  Nombre:          " << e->getNombre()
			  << "\n  " << etiqueta << ": " << e->obtenerDetallePuesto()
			  << "\n  Departamento:    " << nombreDepartamento
			  << "\n  Puesto:          " << textoOSinAsignar(e->getCodigoPuesto())
			  << "\n";

	std::cout << "\n  --- Desglose salarial ---"
			  << "\n  Salario base:                 " << formatearMonto(e->getSalarioBase())
			  << "\n  Horas extra (" << e->getHorasExtra() << " x "
			  << formatearMonto(e->getValorHoraExtra()) << "):  "
			  << formatearMonto(e->calcularPagoHorasExtra())
			  << "\n  Turnos nocturnos (" << e->getTurnosNocturnos() << " x "
			  << formatearMonto(e->getBonificacionTurnoNocturno()) << "):  "
			  << formatearMonto(e->calcularBonificacionNocturna())
			  << "\n  Guardias (" << e->getGuardias() << " x "
			  << formatearMonto(e->getBonificacionGuardia()) << "):  "
			  << formatearMonto(e->calcularBonificacionGuardias())
			  << "\n  Bonificación del puesto:      "
			  << formatearMonto(e->calcularBonificacionPuesto())
			  << "\n  Salario bruto:                " << formatearMonto(e->calcularSalarioBruto())
			  << "\n  Deducción de ley ("
			  << formatearMonto(e->getTasaDeduccionLegal() * 100.0) << "%):  "
			  << formatearMonto(e->calcularDeduccionLegal())
			  << "\n  SALARIO NETO:                 " << formatearMonto(e->calcularSalarioNeto())
			  << "\n";
}

void InterfazConsola::verDetalleEmpleado() {

void InterfazConsola::modificarEmpleado() {
	mostrarTitulo("Modificar empleado");

	Empleado* e = solicitarEmpleadoExistente();
	mostrarEmpleado(e);

	const std::string etiqueta = fabrica.obtenerEtiquetaDetalle(e->obtenerTipo());

	std::cout << "\nCampos modificables:\n"
			  << "   1. Nombre\n"
			  << "   2. Salario base\n"
			  << "   3. Horas extra\n"
			  << "   4. Valor por hora extra\n"
			  << "   5. Turnos nocturnos\n"
			  << "   6. Bonificación por turno nocturno\n"
			  << "   7. Guardias\n"
			  << "   8. Bonificación por guardia\n"
			  << "   9. Tasa de deducción de ley\n"
			  << "  10. " << etiqueta << "\n"
			  << "  11. Bonificación del puesto\n"
			  << "   0. Cancelar\n";

	int campo = EntradaConsola::leerEntero("Campo a modificar: ", 0, 11);

	switch (campo) {
		case 1:
			e->setNombre(EntradaConsola::leerTexto("Nuevo nombre: "));
			break;
		case 2:
			e->setSalarioBase(EntradaConsola::leerDecimal("Nuevo salario base: "));
			break;
		case 3:
			e->setHorasExtra(EntradaConsola::leerDecimal("Nuevas horas extra: "));
			break;
		case 4:
			e->setValorHoraExtra(EntradaConsola::leerDecimal("Nuevo valor por hora extra: "));
			break;
		case 5:
			e->setTurnosNocturnos(EntradaConsola::leerEntero("Nuevos turnos nocturnos: ", 0));
			break;
		case 6:
			e->setBonificacionTurnoNocturno(EntradaConsola::leerDecimal("Nueva bonificación nocturna: "));
			break;
		case 7:
			e->setGuardias(EntradaConsola::leerEntero("Nueva cantidad de guardias: ", 0));
			break;
		case 8:
			e->setBonificacionGuardia(EntradaConsola::leerDecimal("Nueva bonificación por guardia: "));
			break;
		case 9:
			e->setTasaDeduccionLegal(
				EntradaConsola::leerDecimal("Nueva tasa de deducción en % (0-100): ", 0.0, 100.0) / 100.0);
			break;
		case 10:
			e->setDetallePuesto(EntradaConsola::leerTexto("Nuevo valor de " + etiqueta + ": "));
			break;
		case 11:
			e->setBonificacionPuesto(EntradaConsola::leerDecimal("Nueva bonificación del puesto: "));
			break;
		case 0:
			std::cout << "  Operación cancelada.\n";
			return;
	}

	empleados.guardar();
	std::cout << "\n  Empleado actualizado correctamente.\n";
}

void InterfazConsola::asignarPuestoEmpleado() {
	mostrarTitulo("Asignar departamento y puesto");

	Empleado* e = solicitarEmpleadoExistente();

	listarDepartamentos();

	std::string idDepartamento = EntradaConsola::leerTexto(
		"Código del departamento (vacío para quitar la asignación): ", true);

	std::string codigoPuesto;

	if (!idDepartamento.empty()) {
		codigoPuesto = EntradaConsola::leerTexto(
			"Código del puesto (vacío para ninguno): ", true);
	}

	empleados.asignarPuesto(e->getIdentificacion(), idDepartamento, codigoPuesto);
	std::cout << "\n  Asignación actualizada correctamente.\n";
}

void InterfazConsola::eliminarEmpleado() {
	mostrarTitulo("Eliminar empleado");

	Empleado* e = solicitarEmpleadoExistente();
	mostrarEmpleado(e);

	if (!EntradaConsola::confirmar("\n¿Seguro que desea eliminar a este empleado?")) {
		std::cout << "  Operación cancelada.\n";
		return;
	}

	
	const std::string id = e->getIdentificacion();
	planilla.desvincularEmpleado(id);
	empleados.eliminar(id);

	std::cout << "\n  Empleado eliminado. Sus comprobantes se conservan en el historial.\n";
}



void InterfazConsola::crearDepartamento() {
	mostrarTitulo("Crear departamento");

	std::string codigo = EntradaConsola::leerTexto("Código: ");
	std::string nombre = EntradaConsola::leerTexto("Nombre: ");

	departamentos.crear(codigo, nombre);
	std::cout << "\n  Departamento creado correctamente.\n";
}

void InterfazConsola::listarDepartamentos() const {
	mostrarTitulo("Departamentos y puestos");

	const Lista<Departamento>& lista = departamentos.getDepartamentos();

	if (lista.estaVacia()) {
		std::cout << "  No hay departamentos registrados.\n";
		return;
	}

	for (Nodo<Departamento>* nodo = lista.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		const Departamento* d = nodo->getDato();

		std::cout << "\n  [" << d->getCodigo() << "] " << d->getNombre()
				  << "  (" << empleados.contarPorDepartamento(d->getCodigo())
				  << " empleado(s))\n";

		const Lista<Puesto>& puestos = d->getPuestos();

		if (puestos.estaVacia()) {
			std::cout << "      (sin puestos)\n";
		}

		for (Nodo<Puesto>* np = puestos.primero();
			 np != nullptr;
			 np = np->getSiguiente()) {

			std::cout << "      - " << np->getDato()->getCodigo()
					  << ": " << np->getDato()->getNombre() << "\n";
		}
	}
}

void InterfazConsola::agregarPuesto() {
	mostrarTitulo("Agregar puesto");

	std::string codigoDepartamento = EntradaConsola::leerTexto("Código del departamento: ");
	std::string codigoPuesto = EntradaConsola::leerTexto("Código del puesto: ");
	std::string nombrePuesto = EntradaConsola::leerTexto("Nombre del puesto: ");

	departamentos.agregarPuesto(codigoDepartamento, codigoPuesto, nombrePuesto);
	std::cout << "\n  Puesto agregado correctamente.\n";
}

void InterfazConsola::eliminarPuesto() {
	mostrarTitulo("Eliminar puesto");

	std::string codigoDepartamento = EntradaConsola::leerTexto("Código del departamento: ");
	std::string codigoPuesto = EntradaConsola::leerTexto("Código del puesto: ");

	if (!EntradaConsola::confirmar("¿Seguro que desea eliminar el puesto?")) {
		std::cout << "  Operación cancelada.\n";
		return;
	}

	departamentos.eliminarPuesto(codigoDepartamento, codigoPuesto);
	empleados.desasignarPuesto(codigoDepartamento, codigoPuesto);

	std::cout << "\n  Puesto eliminado correctamente.\n";
}

void InterfazConsola::eliminarDepartamento() {
	mostrarTitulo("Eliminar departamento");

	std::string codigo = EntradaConsola::leerTexto("Código del departamento: ");
	const Departamento* d = departamentos.buscar(codigo);

	if (d == nullptr) {
		throw std::invalid_argument(
			"No existe un departamento con el codigo " + codigo + ".");
	}

	int asignados = empleados.contarPorDepartamento(codigo);

	std::cout << "\n  Departamento: " << d->getNombre()
			  << "\n  Empleados asignados que quedarán sin departamento: "
			  << asignados << "\n";

	if (!EntradaConsola::confirmar("\n¿Seguro que desea eliminarlo junto con sus puestos?")) {
		std::cout << "  Operación cancelada.\n";
		return;
	}

	planilla.desvincularDepartamento(codigo);
	empleados.desasignarDepartamento(codigo);
	departamentos.eliminar(codigo);

	std::cout << "\n  Departamento eliminado correctamente.\n";
}



void InterfazConsola::generarComprobanteIndividual() {
	mostrarTitulo("Generar comprobante");

	Empleado* e = solicitarEmpleadoExistente();
	std::string periodo = EntradaConsola::leerPeriodo("Período (AAAA-MM): ");

		planilla.generarComprobante(e->getIdentificacion(), periodo);

	mostrarComprobante(comprobante);
}

void InterfazConsola::generarPlanillaPeriodo() {
	mostrarTitulo("Generar planilla del período");

	if (empleados.getEmpleados().estaVacia()) {
		throw std::logic_error("No hay empleados registrados.");
	}

	std::string periodo = EntradaConsola::leerPeriodo("Período (AAAA-MM): ");
	int generados = planilla.generarPlanilla(periodo);

	std::cout << "\n  Comprobantes generados: " << generados << "\n";

	if (generados == 0) {
		std::cout << "  (Todos los empleados ya tenían comprobante en ese período.)\n";
	}
}

void InterfazConsola::listarComprobantes() const {
	mostrarTitulo("Comprobantes emitidos");

	const Lista<ComprobantePago>& lista = planilla.getComprobantes();

	if (lista.estaVacia()) {
		std::cout << "  No hay comprobantes emitidos.\n";
		return;
	}

	std::cout << std::left
			  << std::setw(8) << "No."
			  << std::setw(10) << "Período"
			  << std::setw(14) << "Empleado"
			  << std::setw(14) << "Neto"
			  << std::setw(11) << "Estado"
			  << "Nombre\n";

	for (Nodo<ComprobantePago>* nodo = lista.primero();
		 nodo != nullptr;
		 nodo = nodo->getSiguiente()) {

		const ComprobantePago* c = nodo->getDato();

		std::cout << std::left
				  << std::setw(8) << c->getNumero()
				  << std::setw(10) << c->getPeriodo()
				  << std::setw(14) << c->getIdEmpleado()
				  << std::setw(14) << formatearMonto(c->getDesglose().salarioNeto)
				  << std::setw(11) << (c->estaPagado() ? "Pagado" : "Pendiente")
				  << c->getNombreEmpleado() << "\n";
	}
}

void InterfazConsola::mostrarComprobante(const ComprobantePago* c) const {
	const DesglosePago& d = c->getDesglose();

	std::cout << "\n  ------------------------------------------"
			  << "\n  COMPROBANTE DE PAGO No. " << c->getNumero()
			  << "\n  ------------------------------------------"
			  << "\n  Período:         " << c->getPeriodo()
			  << "\n  Fecha emisión:   " << c->getFechaEmision()
			  << "\n  Estado:          "
			  << (c->estaPagado() ? "Pagado el " + c->getFechaPago() : "Pendiente de pago")
			  << "\n  Empleado:        " << c->getNombreEmpleado()
			  << " (" << c->getIdEmpleado() << ")"
			  << (c->getEmpleado() == nullptr ? "  [ya no está en el expediente]" : "")
			  << "\n  Tipo:            " << c->getTipoEmpleado()
			  << "\n  Departamento:    "
			  << (c->getCodigoDepartamento().empty()
					  ? "Sin asignar"
					  : c->getCodigoDepartamento() + " - " + c->getNombreDepartamento())
			  << (!c->getCodigoDepartamento().empty() && c->getDepartamento() == nullptr
					  ? "  [ya no existe]" : "")
			  << "\n"
			  << "\n  Salario base:             " << formatearMonto(d.salarioBase)
			  << "\n  Horas extra:              " << formatearMonto(d.pagoHorasExtra)
			  << "\n  Bonificación nocturna:    " << formatearMonto(d.bonificacionNocturna)
			  << "\n  Bonificación guardias:    " << formatearMonto(d.bonificacionGuardias)
			  << "\n  Bonificación del puesto:  " << formatearMonto(d.bonificacionPuesto)
			  << "\n  Salario bruto:            " << formatearMonto(d.salarioBruto)
			  << "\n  Deducción de ley:         " << formatearMonto(d.deduccionLegal)
			  << "\n  SALARIO NETO:             " << formatearMonto(d.salarioNeto)
			  << "\n  ------------------------------------------\n";
}

void InterfazConsola::verComprobante() {
	mostrarTitulo("Ver comprobante");

	int numero = EntradaConsola::leerEntero("Número de comprobante: ", 1);
	const ComprobantePago* comprobante = planilla.buscarComprobante(numero);

	if (comprobante == nullptr) {
		throw std::invalid_argument(
			"No existe el comprobante numero " + std::to_string(numero) + ".");
	}

	mostrarComprobante(comprobante);
}

void InterfazConsola::procesarPagoIndividual() {
	mostrarTitulo("Procesar pago");

	int numero = EntradaConsola::leerEntero("Número de comprobante: ", 1);
	const ComprobantePago* comprobante = planilla.buscarComprobante(numero);

	if (comprobante == nullptr) {
		throw std::invalid_argument(
			"No existe el comprobante numero " + std::to_string(numero) + ".");
	}

	mostrarComprobante(comprobante);

	if (!EntradaConsola::confirmar("\n¿Confirma el pago de este comprobante?")) {
		std::cout << "  Operación cancelada.\n";
		return;
	}

	planilla.procesarPago(numero);
	std::cout << "\n  Pago procesado correctamente.\n";
}

void InterfazConsola::procesarPagosDelPeriodo() {
	mostrarTitulo("Procesar pagos del período");

	std::string periodo = EntradaConsola::leerPeriodo("Período (AAAA-MM): ");

	if (!EntradaConsola::confirmar("¿Confirma el pago de todos los comprobantes pendientes?")) {
		std::cout << "  Operación cancelada.\n";
		return;
	}

	int pagados = planilla.procesarPagosPeriodo(periodo);
	std::cout << "\n  Comprobantes pagados: " << pagados << "\n";
}

void InterfazConsola::mostrarResumenPeriodo() {
	mostrarTitulo("Resumen del período");

	std::string periodo = EntradaConsola::leerPeriodo("Período (AAAA-MM): ");
	int cantidad = planilla.contarComprobantes(periodo);

	if (cantidad == 0) {
		std::cout << "  No hay comprobantes en el período " << periodo << ".\n";
		return;
	}

	std::cout << "\n  Período:                " << periodo
			  << "\n  Comprobantes emitidos:  " << cantidad
			  << "\n  Total salario bruto:    " << formatearMonto(planilla.calcularTotalBruto(periodo))
			  << "\n  Total deducciones:      " << formatearMonto(planilla.calcularTotalDeducciones(periodo))
			  << "\n  Total salario neto:     " << formatearMonto(planilla.calcularTotalNeto(periodo))
			  << "\n";
}
